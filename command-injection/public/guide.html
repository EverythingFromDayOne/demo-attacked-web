; or & — command separator: runs next command regardless of first result
&& — runs next command only if first succeeds
| — pipes stdout of first command as stdin to second
$(...) — command substitution: output replaces the expression

// SAFE: execFile() takes command + args as separate array // Shell is never invoked — metacharacters are just text const { execFile } = require('child_process'); execFile('ping', ['-n', '4', hostname], callback); // hostname = "localhost & whoami" // ping receives: ["-n", "4", "localhost & whoami"] // The & is just a string argument — ping tries to resolve it
